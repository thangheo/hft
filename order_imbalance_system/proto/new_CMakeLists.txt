# proto/CMakeLists.txt

# Define Protobuf and gRPC generated sources
set(PROTO_FILES
    market_data.proto
    order_book.proto
    execution_service.proto
)

# Create custom command for protobuf code generation
set(PROTO_SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR})
set(PROTO_BINARY_DIR ${CMAKE_CURRENT_BINARY_DIR})

foreach(proto ${PROTO_FILES})
    get_filename_component(proto_name ${proto} NAME_WE)
    list(APPEND PROTO_SRCS "${PROTO_BINARY_DIR}/${proto_name}.pb.cc")
    list(APPEND PROTO_HDRS "${PROTO_BINARY_DIR}/${proto_name}.pb.h")
    list(APPEND GRPC_SRCS "${PROTO_BINARY_DIR}/${proto_name}.grpc.pb.cc")
    list(APPEND GRPC_HDRS "${PROTO_BINARY_DIR}/${proto_name}.grpc.pb.h")

    add_custom_command(
        OUTPUT "${PROTO_BINARY_DIR}/${proto_name}.pb.cc"
               "${PROTO_BINARY_DIR}/${proto_name}.pb.h"
               "${PROTO_BINARY_DIR}/${proto_name}.grpc.pb.cc"
               "${PROTO_BINARY_DIR}/${proto_name}.grpc.pb.h"
        COMMAND ${PROTOBUF_PROTOC_EXECUTABLE}
        ARGS --grpc_out=${PROTO_BINARY_DIR}
             --cpp_out=${PROTO_BINARY_DIR}
             --proto_path=${PROTO_SRC_DIR}
             --plugin=protoc-gen-grpc=${GRPC_CPP_PLUGIN_EXECUTABLE}
             ${PROTO_SRC_DIR}/${proto}
        DEPENDS ${PROTO_SRC_DIR}/${proto}
        COMMENT "Running protoc and grpc_cpp_plugin on ${proto}"
        VERBATIM
    )
endforeach()

# Add a library target
add_library(proto_libs ${PROTO_SRCS} ${PROTO_HDRS} ${GRPC_SRCS} ${GRPC_HDRS})

# Set include directories
target_include_directories(proto_libs
    PUBLIC
        ${CMAKE_CURRENT_BINARY_DIR}
        ${Protobuf_INCLUDE_DIRS}
        ${gRPC_INCLUDE_DIRS}
)

# Link libraries
target_link_libraries(proto_libs
    PUBLIC
        ${Protobuf_LIBRARIES}
        ${gRPC_LIBRARIES}
)
